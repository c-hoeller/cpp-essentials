Templates (nur C++) erlauben generischen Code, der mit verschiedenen Datentypen funktioniert, ohne ihn mehrfach zu schreiben.

Funktionstemplate

C++
template <typename T>
T maximum(T a, T b) {
    return (a > b) ? a : b;
}

cout << maximum(3, 7) << endl;       // T = int    -> 7
cout << maximum(3.5, 2.1) << endl;   // T = double -> 3.5

Klassentemplate

C++
template <typename T>
class Box {
private:
    T inhalt;
public:
    Box(T wert) : inhalt(wert) {}
    T getInhalt() const { return inhalt; }
};

Box<int> b1(42);
Box<string> b2("Hallo");

std::vector (STL)

C++
#include <vector>

vector<int> zahlen = {1, 2, 3};
zahlen.push_back(4);          // am Ende anhaengen -> {1,2,3,4}
cout << zahlen.size() << endl; // 4
cout << zahlen[1] << endl;     // 2 (Index-Zugriff wie beim Array)

for (int n : zahlen) {         // range-based for (C++11)
    cout << n << " ";
}

vector ist ein dynamisches Array, das automatisch wächst – im Gegensatz zu C-Arrays muss die Größe nicht im Voraus feststehen.
