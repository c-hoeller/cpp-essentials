Lesen Sie eine Anzahl Sekunden ein und rechnen Sie sie in Stunden, Minuten (max. 59) und Sekunden (max. 59) um. Geben Sie die drei Werte jeweils in einer eigenen Zeile aus, ohne erläuternden Text.

Beispiele

EingabeAusgabe
450 0 45
1300 2 10
73002 1 40

Lösung

C++
#include <iostream>
#include <cstdlib>
using namespace std;

int main()
{
    int seconds;
    cin >> seconds;

    int hour_seconds = 3600; // 60*60 seconds
    int minute_seconds = 60; // 60 seconds

    int hours = seconds / hour_seconds;
    int rest_seconds = seconds - (hours * hour_seconds);

    int minutes = rest_seconds / minute_seconds;
    rest_seconds = rest_seconds - (minutes * minute_seconds);

    cout << hours << endl;
    cout << minutes << endl;
    cout << rest_seconds << endl;

    return 0;
}
