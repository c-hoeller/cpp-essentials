Der Benutzer gibt eine Uhrzeit im Format hh:mm ein (beide Teile immer zweistellig) sowie eine Anzahl Minuten. Berechnen und geben Sie die Uhrzeit aus, die sich ergibt, wenn man diese Minutenanzahl von der eingegebenen Uhrzeit zurückrechnet (ebenfalls im Format hh:mm). Ein Tageswechsel muss nicht berücksichtigt werden.

Beispiele

EingabeAusgabe
12:34 2012:14
12:34 6011:34
12:34 18109:33

Lösung

C++
#include <iostream>
#include <cstdlib>
#include <string>
using namespace std;

int main()
{
    string uhrzeit;
    int minutenabzug;
    cin >> uhrzeit >> minutenabzug;

    int stunden = stoi(uhrzeit.substr(0, 2));
    int minuten = stoi(uhrzeit.substr(3, 2));

    int gesamtminuten = stunden * 60 + minuten - minutenabzug;

    int ergebnis_stunden = gesamtminuten / 60;
    int ergebnis_minuten = gesamtminuten % 60;

    if (ergebnis_stunden < 10)
        cout << "0" << ergebnis_stunden;
    else
        cout << ergebnis_stunden;

    cout << ":";

    if (ergebnis_minuten < 10)
        cout << "0" << ergebnis_minuten;
    else
        cout << ergebnis_minuten;

    cout << endl;

    return 0;
}
