Initialisieren Sie ein zweidimensionales Integer-Array mit folgenden Werten:

Beispiel
2  5  7  9  2  4
1  7  3  7  9  2
3  6  8  2  5  8

Der Benutzer gibt eine Zeilen- und eine Spaltennummer ein (Zählung ab 0). Ausgegeben wird der Wert an dieser Position, bzw. -1, wenn die Position außerhalb des Arrays liegt.

Beispiele

EingabeAusgabe
4 4-1
2 28
1 37

Lösung

C++
#include <iostream>

using namespace std;

int main()
{
    int arr[3][6] = {{2, 5, 7, 9, 2, 4},
                     {1, 7, 3, 7, 9, 2},
                     {3, 6, 8, 2, 5, 8}};

    int zeile, spalte;
    cin >> zeile >> spalte;

    if (zeile < 0 || zeile >= 3 || spalte < 0 || spalte >= 6)
    {
        cout << -1 << endl;
    }
    else
    {
        cout << arr[zeile][spalte] << endl;
    }

    return 0;
}
