Eine abgeleitete Klasse erbt Attribute und Methoden einer Basisklasse und kann sie erweitern oder überschreiben.

C++
class Tier {
public:
    virtual void sagHallo() const {
        cout << "Ich bin ein Tier" << endl;
    }
    virtual ~Tier() {}   // virtueller Destruktor: wichtig bei Vererbung + delete ueber Basisklassenzeiger
};

class Hund : public Tier {     // : public Basisklasse
public:
    void sagHallo() const override {  // override: ueberschreibt die Basisversion
        cout << "Wuff!" << endl;
    }
};

int main() {
    Tier* t = new Hund();
    t->sagHallo();     // "Wuff!" – dank virtual wird die Hund-Version aufgerufen
    delete t;
}
Ohne virtual in der Basisklasse würde t->sagHallo() trotz eines Hund-Objekts die Tier-Version aufrufen (statische statt dynamische Bindung) – ein häufiger Klausur-Stolperstein.

Abstrakte Klassen

C++
class Form {
public:
    virtual double flaeche() const = 0;   // rein virtuell -> Form ist abstrakt
};
// Form f;           // FEHLER: abstrakte Klassen koennen nicht instanziiert werden

class Kreis : public Form {
private:
    double radius;
public:
    Kreis(double r) : radius(r) {}
    double flaeche() const override { return 3.14159 * radius * radius; }
};

Basisklassenkonstruktor aufrufen

Hat die Basisklasse einen Konstruktor mit Parametern, muss die abgeleitete Klasse ihn explizit in ihrer eigenen Initialisierungsliste aufrufen, bevor sie ihre eigenen Attribute initialisiert:

C++
class Ball {
protected:
    double v0;      // Abschussgeschwindigkeit
    double beta;    // Abschusswinkel
public:
    Ball(double v0, double beta) : v0(v0), beta(beta) {}
    virtual void flugbahn() const { cout << "einfache Flugbahn" << endl; }
};

class Golfball : public Ball {
private:
    int anzahlDimples;
public:
    // Basisklassen-Konstruktor wird explizit in der Initialisierungsliste aufgerufen:
    Golfball(double v0, double beta, int dimples)
        : Ball(v0, beta), anzahlDimples(dimples) {}

    void flugbahn() const override {
        cout << "Flugbahn mit " << anzahlDimples << " Dimples" << endl;
    }
};
